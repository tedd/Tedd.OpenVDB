cmake_minimum_required(VERSION 3.28)
project(NativeCacheTransfer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
foreach(variant stock cache mask both split split_both split_only)
  add_executable(${variant} main.cpp)
  if(NOT variant STREQUAL "stock")
    target_include_directories(${variant} BEFORE PRIVATE "${PATCH_ROOT}")
  endif()
  target_include_directories(${variant} PRIVATE "${OPENVDB_SOURCE_DIR}/openvdb"
    "${OPENVDB_SOURCE_DIR}/ext/imath" "${OPENVDB_BUILD_DIR}/openvdb/openvdb"
    "${OPENVDB_BUILD_DIR}/openvdb/openvdb/openvdb" "${TBB_ROOT}/include")
  target_compile_definitions(${variant} PRIVATE OPENVDB_STATICLIB NOMINMAX WIN32_LEAN_AND_MEAN)
  if(variant STREQUAL "cache" OR variant STREQUAL "both" OR variant STREQUAL "split" OR variant STREQUAL "split_both")
    target_compile_definitions(${variant} PRIVATE EXPERIMENT_LEAF_CACHE)
  endif()
  if(variant STREQUAL "mask" OR variant STREQUAL "both" OR variant STREQUAL "split_both")
    target_compile_definitions(${variant} PRIVATE EXPERIMENT_DENSE_MASK)
  endif()
  if(variant STREQUAL "split" OR variant STREQUAL "split_both" OR variant STREQUAL "split_only")
    target_compile_definitions(${variant} PRIVATE EXPERIMENT_SPLIT_CACHE)
  endif()
  target_compile_options(${variant} PRIVATE /FAs "/Fa${CMAKE_CURRENT_BINARY_DIR}/${variant}.asm")
  target_link_libraries(${variant} PRIVATE "${OPENVDB_BUILD_DIR}/openvdb/openvdb/Release/libopenvdb.lib"
    "${TBB_ROOT}/lib/intel64/vc14/tbb12.lib" "${ZLIB_LIBRARY}")
  set_property(TARGET ${variant} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endforeach()
