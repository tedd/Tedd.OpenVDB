cmake_minimum_required(VERSION 3.28)
project(TeddAdvectionReference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)

set_property(TARGET levelset-advection-reference PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(levelset-advection-reference PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/bigobj>)
