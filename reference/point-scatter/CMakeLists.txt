cmake_minimum_required(VERSION 3.28)
project(TeddPointScatterReference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(point-scatter-reference ../point-scatter-oracle.cpp)
target_include_directories(point-scatter-reference PRIVATE "${OPENVDB_SOURCE_DIR}/openvdb" "${OPENVDB_SOURCE_DIR}/ext/imath" "${OPENVDB_BUILD_DIR}/openvdb/openvdb" "${OPENVDB_BUILD_DIR}/openvdb/openvdb/openvdb" "${TBB_ROOT}/include")
target_compile_definitions(point-scatter-reference PRIVATE OPENVDB_STATICLIB _USE_MATH_DEFINES NOMINMAX)
target_link_libraries(point-scatter-reference PRIVATE "${OPENVDB_BUILD_DIR}/openvdb/openvdb/Release/libopenvdb.lib" "${TBB_ROOT}/lib/intel64/vc14/tbb12.lib")
set_property(TARGET point-scatter-reference PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(point-scatter-reference PRIVATE /bigobj)
