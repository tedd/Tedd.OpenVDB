cmake_minimum_required(VERSION 3.28)
project(TeddRayTracerReference LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(ray-tracer-reference ../ray-tracer-oracle.cpp)
set_source_files_properties(../ray-tracer-oracle.cpp PROPERTIES LANGUAGE CXX)
target_include_directories(ray-tracer-reference PRIVATE "${OPENVDB_SOURCE_DIR}/openvdb" "${OPENVDB_SOURCE_DIR}/ext/imath" "${OPENVDB_BUILD_DIR}/openvdb/openvdb" "${OPENVDB_BUILD_DIR}/openvdb/openvdb/openvdb" "${TBB_ROOT}/include")
target_compile_definitions(ray-tracer-reference PRIVATE OPENVDB_STATICLIB _USE_MATH_DEFINES NOMINMAX)
target_link_libraries(ray-tracer-reference PRIVATE "${OPENVDB_BUILD_DIR}/openvdb/openvdb/Release/libopenvdb.lib" "${TBB_ROOT}/lib/intel64/vc14/tbb12.lib")
set_property(TARGET ray-tracer-reference PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_compile_options(ray-tracer-reference PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/bigobj>)
