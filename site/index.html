Program.csC# / .NET 11
using Tedd.OpenVDB.Math;
using Tedd.OpenVDB.Tree;

// Empty space returns the background value.
var tree = new VdbTree<ushort>(background: 0);
var region = new CoordBBox(
    new(-32, -32, -32), new(-1, -1, -1));

tree.Fill(region, 1);
var accessor = tree.GetAccessor();

// Edit one voxel inside the region.
accessor.SetValueOn(new(-16, -16, -16), 2);

// Deactivate it; the value is still 2.
accessor.SetValueOff(new(-16, -16, -16));
tree.Prune();
